cmake_minimum_required(VERSION 3.24...4.4)
project(colorist_fill_bench LANGUAGES CXX)
find_package(OpenCV 4.14 REQUIRED COMPONENTS core imgcodecs)
add_executable(fill_bench main.cpp)
set_target_properties(fill_bench PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(fill_bench PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(fill_bench PRIVATE ${OpenCV_LIBS})
include(../../cmake/simd.cmake)
colorist_select_simd(fill_bench)
if(MINGW)
    target_link_options(fill_bench PRIVATE -static)
endif()
